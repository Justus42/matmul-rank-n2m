## 2. Methoden

### 2.1 Flip-Graph-Wanderung über GF(p)

Eigenes C-Programm (Anhang A). Ein Verfahren ist eine Liste von Termen u⊗v⊗w mit u ∈ GF(p)^(4·2) (Koeffizienten auf A), v ∈ GF(p)^(2·5) (auf B), w ∈ GF(p)^(4·5) (Beitrag zu C = A·B). Gültigkeit: Σ_r u_r[(i,j)]·v_r[(j',k)]·w_r[(i',k')] = δ(i=i')·δ(j=j')·δ(k=k') für alle Indizes.

- **Flip:** zwei Terme mit kollinearem Faktor, etwa u_r = λ·u_s. Dann wird w_r ← w_r + μ·w_s und v_s ← v_s − λμ·v_r (μ ∈ GF(p)^× zufällig; die Rollen der beiden anderen Faktoren werden zufällig vertauscht). Die Summe der beiden Terme bleibt gleich.
- **Reduktion:** stimmen zwei Terme in zwei Faktoren bis auf Skalare überein, werden sie zu einem Term verschmolzen; Terme mit Nullfaktor fallen weg. Nach jedem Flip wird auf Reduktionen geprüft.
- **Plus-Übergang:** nach `plateau` Flips ohne Reduktion wird ein Term in zwei zerlegt (a⊗b⊗c + a′⊗b′⊗c′ = (a−a′)⊗b⊗c + a′⊗b⊗c + a′⊗b′⊗c′), solange der Rang höchstens `maxup` über dem besten liegt; sonst Rücksprung zum besten Verfahren. Mit `maxup` = 0 entspricht das Neustarts vom besten Verfahren.
- Jedes neue beste Verfahren wird im Programm exakt modulo p nachgerechnet (Abbruch bei Fehler; das trat nie ein), gesammelte Verfahren zusätzlich mit einem unabhängigen Python-Skript (Anhang B).

// A is N x M, B is M x P, C = A*B is N x P.  Term: (u.A)*(v.B) added with weight w to C.
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>
#include <time.h>
#define N 4
#define M 2
#define P 5
#define DU (N*M)
#define DV (M*P)
#define DW (N*P)
#define MAXR 64
static int p;
static int R;
typedef struct { uint8_t f[3][DW]; uint64_t h[3]; } Term;
static Term T[MAXR];
static const int dim[3]={DU,DV,DW};
static uint64_t rs;
static inline uint64_t rnd(void){ rs^=rs<<13; rs^=rs>>7; rs^=rs<<17; return rs; }
static int inv[16];
static uint64_t canon_hash(const uint8_t*x,int d){ // hash of x scaled so first nonzero is 1; 0 for zero vector
  int i=0; while(i<d && !x[i]) i++; if(i==d) return 0;
  int s=inv[x[i]]; uint64_t h=1469598103934665603ULL;
  for(;i<d;i++){ h^=(x[i]*s)%p + 1 + 7*i; h*=1099511628211ULL; } return h|1; }
static int scal(const uint8_t*x,const uint8_t*y,int d){ // returns l with x = l*y (assumes collinear, y nonzero)
  for(int i=0;i<d;i++) if(y[i]) return (x[i]*inv[y[i]])%p; return 0; }
static void rehash(int r){ for(int t=0;t<3;t++) T[r].h[t]=canon_hash(T[r].f[t],dim[t]); }
static void delterm(int r){ T[r]=T[R-1]; R--; }
// try reductions; returns 1 if something reduced
static int reduce_all(void){
  int changed=1, any=0;
  while(changed){ changed=0;
    for(int r=0;r<R && !changed;r++){
      for(int t=0;t<3;t++) if(T[r].h[t]==0){ delterm(r); changed=1; any=1; break; }
      if(changed) break;
      for(int q=r+1;q<R && !changed;q++){

          rehash(q); delterm(r); changed=1; any=1; }
      }
    }
  }
  return any;
}
static void init_std(void){ R=0; memset(T,0,sizeof(T));
  for(int i=0;i<N;i++)for(int j=0;j<M;j++)for(int k=0;k<P;k++){ Term*t=&T[R++];

// flip: pick random pair sharing factor t (collinear). returns 1 if done
static int flip(void){
  for(int tries=0;tries<200;tries++){
    int r=rnd()%R, t=rnd()%3; int cand[MAXR],nc=0;
    for(int q=0;q<R;q++) if(q!=r && T[q].h[t]==T[r].h[t]) cand[nc++]=q;
    if(!nc) continue; int s=cand[rnd()%nc];
    // u_r = lam u_s  (t is shared). other two factors: x=(t+1)%3, y=(t+2)%3 ; randomly swap
    int x=(t+1)%3,y=(t+2)%3; if(rnd()&1){int z=x;x=y;y=z;}
    int lam=scal(T[r].f[t],T[s].f[t],dim[t]); int mu=1+rnd()%(p-1);

    for(int i=0;i<dim[y];i++) T[r].f[y][i]=(T[r].f[y][i]+mu*T[s].f[y][i])%p;

    rehash(r); rehash(s); return 1; }
  return 0; }
// plus transition: take two terms r,s with differing t-factor: a r + a' s -> (a-a') r_bc + a' (b c + b' c')
static void plus(void){
  int r=rnd()%R, s; do s=rnd()%R; while(s==r);
  int t=rnd()%3; if(R>=MAXR) return;
  Term nt=T[s]; // a' ⊗ b ⊗ c  where b,c from r
  for(int k=0;k<3;k++) if(k!=t) memcpy(nt.f[k],T[r].f[k],dim[k]);

  T[R++]=nt; rehash(r); rehash(R-1); }

# Exact verification of a <4,2,5> scheme. Usage: python3 verify.py file [p]
# File format: first line "p=.. R=..", then R lines "u v w" with digit strings (entries mod p).
import sys
from fractions import Fraction
N,M,P=4,2,5
def load(fn):
    L=open(fn).read().split('\n')
    hdr=L[0]; p=int(hdr.split()[0][2:]); R=int(hdr.split()[1][2:])
    terms=[[list(map(int,s)) for s in l.split()] for l in L[1:1+R]]
    return p,terms
def check(terms,mod=None):
    for a in range(N*M):
        for b in range(M*P):
            for c in range(N*P):
                s=sum(t[0][a]*t[1][b]*t[2][c] for t in terms)
                i,j=divmod(a,M); j2,k=divmod(b,P); i2,k2=divmod(c,P)
                want=int(i==i2 and j==j2 and k==k2)
                d=s-want
                if (d%mod if mod else d)!=0: return False
    return True
if __name__=='__main__':
    p,terms=load(sys.argv[1])
    print('terms',len(terms),'p',p,'valid mod p:',check(terms,p))
    if p==3:

# Try to lift a GF(2) scheme of <4,2,5> to Z with entries in {-1,0,1} (same support) via SAT.
import sys, itertools
from pysat.formula import IDPool
from pysat.card import CardEnc, EncType
from pysat.solvers import Solver
from verify import load, check, N, M, P
from hensel import target, DIM  # siehe Hilfsteil unten
def signlift(terms, timeout_conf=None):
    pool=IDPool(); cl=[]
    var=lambda r,t,i: pool.id(('x',r,t,i))
    R=len(terms)
    # gauge: in each term, first nonzero entry of factor 0 and factor 1 positive
    for r in range(R):
        for t in (0,1):
            i=next(i for i,v in enumerate(terms[r][t]) if v)
            cl.append([-var(r,t,i)])
    ycache={}
    def y(r,a,b,c):
        key=(r,a,b,c)
        if key in ycache: return ycache[key]
        v=pool.id(('y',)+key); x1,x2,x3=var(r,0,a),var(r,1,b),var(r,2,c)
        for s in itertools.product((0,1),repeat=3):
            lits=[ (x if bit==0 else -x) for x,bit in zip((x1,x2,x3),s)]
            par=sum(s)%2  # assignment where x_i = s_i ; clause excludes assignment s with y != parity
            cl.append(lits+([v] if par else [-v]))
        ycache[key]=v; return v
    for a in range(DIM[0]):
        for b in range(DIM[1]):
            for c in range(DIM[2]):
                S=[r for r in range(R) if terms[r][0][a] and terms[r][1][b] and terms[r][2][c]]
                tg=target(a,b,c)
                if (len(S)-tg)%2: return 'invalid'
                k=(len(S)-tg)//2
                if k<0: return 'unsat'
                if not S: continue
                ys=[y(r,a,b,c) for r in S]
                if k==0: cl.extend([[-v] for v in ys]); continue
                if k==len(S): cl.extend([[v] for v in ys]); continue
                enc=CardEnc.equals(lits=ys,bound=k,vpool=pool,encoding=EncType.seqcounter)
                cl.extend(enc.clauses)
    with Solver(name='cadical153',bootstrap_with=cl) as s:
        if not s.solve(): return 'unsat'
        model=set(l for l in s.get_model() if l>0)
    Z=[[[ (0 if not terms[r][t][i] else (-1 if var(r,t,i) in model else 1)) for i in range(DIM[t])] for t in range(3)] for r in range(R)]
    return Z
if __name__=='__main__':
